/**
 * Non-idempotent requests must never be auto-retried on timeout/network error —
 * a timed-out POST may have been processed server-side (e.g. verification email
 * already sent); retrying duplicates the mutation.
 */
import { httpPost, httpGet } from '../http';

describe('http retry idempotency', () => {
  afterEach(() => jest.restoreAllMocks());

  it('does NOT retry a POST that times out', async () => {
    const abortErr: any = new Error('Aborted');
    abortErr.name = 'AbortError';
    const fetchMock = jest.spyOn(global, 'fetch' as any).mockRejectedValue(abortErr);
    await expect(httpPost('/auth/request-verification', {})).rejects.toBeTruthy();
    expect(fetchMock).toHaveBeenCalledTimes(1);
  });

  it('does NOT retry a POST on a network error', async () => {
    const fetchMock = jest
      .spyOn(global, 'fetch' as any)
      .mockRejectedValue(new Error('Network request failed'));
    await expect(httpPost('/posts', {})).rejects.toBeTruthy();
    expect(fetchMock).toHaveBeenCalledTimes(1);
  });

  it('still retries a GET that times out', async () => {
    const abortErr: any = new Error('Aborted');
    abortErr.name = 'AbortError';
    const fetchMock = jest.spyOn(global, 'fetch' as any).mockRejectedValue(abortErr);
    await expect(httpGet('/highlights')).rejects.toBeTruthy();
    expect(fetchMock.mock.calls.length).toBeGreaterThan(1);
  });
});
```

// A Game row is only a competitive matchup when an opponent actually exists.
// Non-competitive events (fundraiser, watch party, team trip, meeting, meal,
// other) are Game rows too — they must show their own title, never "vs TBD".
export function gameRowTitle(g: {
  title?: string | null;
  event_type?: string | null;
  away_team?: string | null;
  away_team_name?: string | null;
  opponent?: string | null;
}): string {
  const opponent = (g.opponent || g.away_team || g.away_team_name || '').trim();
  const isCompetitive = !g.event_type || g.event_type === 'game' ? true : false;
  if (isCompetitive) return `vs ${opponent || 'TBD'}`;
  return (g.title || '').trim() || eventTypeLabel(g.event_type);
}

function eventTypeLabel(t?: string | null): string {
  switch (t) {
    case 'fundraiser':
      return 'Fundraiser';
    case 'watch_party':
      return 'Watch Party';
    case 'team_trip':
      return 'Team Trip';
    case 'meeting':
      return 'Meeting';
    case 'team_meal':
      return 'Team Meal';
    default:
      return 'Event';
  }
}
```
